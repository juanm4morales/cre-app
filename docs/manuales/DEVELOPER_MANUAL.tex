\def\markdownLaTeXReadAlignments#1{%
  \advance\markdownLaTeXColumnCounter by 1\relax
  \ifnum\markdownLaTeXColumnTotal>3\relax
    \addto@hook\markdownLaTeXTableAlignment{p{\dimexpr\linewidth/\markdownLaTeXColumnTotal-2\tabcolsep\relax}}%
  \else\ifnum\markdownLaTeXColumnTotal=2\relax
    \ifnum\markdownLaTeXColumnCounter=1\relax
      \addto@hook\markdownLaTeXTableAlignment{p{.24\linewidth}}%
    \else
      \addto@hook\markdownLaTeXTableAlignment{p{.70\linewidth}}%
    \fi
  \else
    \ifnum\markdownLaTeXColumnCounter=1\relax
      \addto@hook\markdownLaTeXTableAlignment{p{.18\linewidth}}%
    \else
      \addto@hook\markdownLaTeXTableAlignment{p{.37\linewidth}}%
    \fi
  \fi\fi
  \ifnum\markdownLaTeXColumnCounter<\markdownLaTeXColumnTotal\relax\else
    \expandafter\@gobble
  \fi\markdownLaTeXReadAlignments}
