\node[staticNode] (statics) at (11.0, 1.8) {

\draw[dbEdge] (gunicorn) -- node[above=2pt,midway,font=\scriptsize,text=dbStroke] {TCP local} (postgres);

% PostgreSQL -> Backup
\draw[edge,dashed] (postgres) -- node[right=3pt,midway,font=\scriptsize] {pg\_dump} (backup);
